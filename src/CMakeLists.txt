cmake_minimum_required(VERSION 3.16)
project(vio_application LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

find_package(Eigen3 REQUIRED NO_MODULE)
add_library(vio_core estimate.cpp frames.cpp)
target_include_directories(vio_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(vio_core PUBLIC Eigen3::Eigen)
target_compile_options(vio_core PRIVATE -Wall -Wextra -Wpedantic)

include(cmake/Mavlink.cmake)
add_library(vio_flight_link flight_session.cpp serial_port.cpp mavlink_link.cpp)
target_link_libraries(vio_flight_link PUBLIC vio_core vio_mavlink_headers)
target_compile_options(vio_flight_link PRIVATE -Wall -Wextra -Wpedantic)

find_package(nlohmann_json 3.11 REQUIRED)
add_library(vio_imu_device imu_device.cpp)
target_include_directories(vio_imu_device PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(vio_imu_device PUBLIC nlohmann_json::nlohmann_json)
target_compile_options(vio_imu_device PRIVATE -Wall -Wextra -Wpedantic)

find_package(Threads REQUIRED)
add_library(vio_recording recording.cpp)
target_include_directories(vio_recording PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(vio_recording PUBLIC Threads::Threads)
target_compile_options(vio_recording PRIVATE -Wall -Wextra -Wpedantic)
add_library(vio_runtime session_store.cpp flight_config.cpp camera_capture.cpp flight_application.cpp)
target_link_libraries(vio_runtime PUBLIC vio_imu_device vio_flight_link vio_recording Threads::Threads)
target_compile_options(vio_runtime PRIVATE -Wall -Wextra -Wpedantic)

include(CTest)
if(BUILD_TESTING)

  target_link_libraries(test_recording PRIVATE vio_recording nlohmann_json::nlohmann_json)

  find_package(PkgConfig REQUIRED)
  pkg_check_modules(LIBCAMERA REQUIRED IMPORTED_TARGET libcamera>=0.4)
  add_library(vio_camera libcamera_source.cpp)
  target_link_libraries(vio_camera PUBLIC vio_runtime PkgConfig::LIBCAMERA)
  target_compile_options(vio_camera PRIVATE -Wall -Wextra -Wpedantic)
  add_subdirectory(openvins_runner)
  add_executable(vio_flight flight_main.cpp)
  target_link_libraries(vio_flight PRIVATE vio_runtime vio_openvins vio_camera)
  if(BUILD_TESTING)
    add_executable(test_estimator_runner ../tests/test_estimator_runner.cpp)
    target_link_libraries(test_estimator_runner PRIVATE vio_runtime vio_openvins)
    add_test(NAME estimator_runner COMMAND test_estimator_runner
      ${CMAKE_CURRENT_SOURCE_DIR}/config/estimator_config.yaml)
    set_tests_properties(estimator_runner PROPERTIES TIMEOUT 20)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME flight_main_signals COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/../tests/test_flight_main.py
      $<TARGET_FILE:vio_flight> ${CMAKE_CURRENT_SOURCE_DIR}/config/flight.json)
    set_tests_properties(flight_main_signals PROPERTIES TIMEOUT 40)
  endif()
